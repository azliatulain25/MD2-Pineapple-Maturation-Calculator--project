    <script>
        // State parameters matching initial sliders
        let L = 1800; // Carrying Capacity (grams)
        let k = 0.40; // Growth coefficient (relative speed)
        let x0 = 10;  // Growth midpoint inflection (weeks)

        // Generate data arrays over a 20 week timeline with 0.5 step accuracy
        const totalWeeks = 20;
        const steps = 40; // 0 to 20 with 0.5 steps
        let selectedIndex = 20; // Default viewing week: Week 10

        // Sigmoidal growth function f(x)
        function f(x, currentL, currentK, currentX0) {
            return currentL / (1 + Math.exp(-currentK * (x - currentX0)));
        }

        // Mathematical Derivative function f'(x)
        function fPrime(x, currentL, currentK, currentX0) {
            const expTerm = Math.exp(-currentK * (x - currentX0));
            const denominator = Math.pow(1 + expTerm, 2);
            return (currentL * currentK * expTerm) / denominator;
        }

        // Generate dynamic dataset array for chart updates
        function generateChartData() {
            const labels = [];
            const dataFx = [];
            const dataFprime = [];

            for (let i = 0; i <= steps; i++) {
                const x = i * 0.5;
                labels.push(x);
                dataFx.push(f(x, L, k, x0).toFixed(1));
                dataFprime.push(fPrime(x, L, k, x0).toFixed(1));
            }
            return { labels, dataFx, dataFprime };
        }

        // Layman descriptive bioscience evaluator
        function getLaymanAssessment(week, weight, rateOfChange, currentK, currentX0) {
            const inflectionThreshold = currentX0;
            const diffFromInflection = week - inflectionThreshold;
            
            if (week <= 4.0) {
                return `**Early Vegetative Induction:** At Week ${week.toFixed(1)}, the MD2 pineapple is in its post-flowering cell expansion phase. Fruit weight is relatively low (${weight.toFixed(0)}g), with slow weight accumulation running at ${rateOfChange.toFixed(1)}g per week. Nitrogen absorption is highly critical here to develop chloroplasts in the crown, though cell division is just gathering speed.`;
            } else if (week > 4.0 && week < (inflectionThreshold - 1)) {
                return `**Pre-Grand Growth Stage:** At Week ${week.toFixed(1)}, growth speed is actively climbing to ${rateOfChange.toFixed(1)}g per week. The crop mass is steadily building (${weight.toFixed(0)}g). It is crucial to administer potassium sulfate during this phase as water uptake is accelerating and structural tissues are hardening rapidly.`;
            } else if (Math.abs(diffFromInflection) <= 1.0) {
                return `**Inflection Point / Peak Mass Intake:** Outstanding! Week ${week.toFixed(1)} marks the absolute grand maturation stage of the MD2 fruit. Weight gain rate is hitting its theoretical limit of ${rateOfChange.toFixed(1)}g per week. Your pineapple has reached roughly half its potential harvest mass (${weight.toFixed(0)}g). Cellular metabolism is running full tilt. Irrigation must remain highly consistent to satisfy physiological pressure.`;
            } else if (week >= (inflectionThreshold + 1) && week <= 15.0) {
                return `**Post-Inflection Plateau:** Inside Week ${week.toFixed(1)}, overall mass is quite heavy (${weight.toFixed(0)}g), but the instantaneous growth speed has down-shifted to ${rateOfChange.toFixed(1)}g per week. Physical size accumulation is gradually finishing. The fruit is pivoting metabolic energy toward standard starch synthesis and sweetness maturation. Keep fertilizer inputs minimal to prevent tissue burn.`;
            } else {
                return `**Senescence / Ripening Stage:** At Week ${week.toFixed(1)}, total weight is capping near maximum limits (${weight.toFixed(0)}g). Growth acceleration is flatlining near ${rateOfChange.toFixed(1)}g per week. MD2 skin is turning golden. This low derivative signifies that ripening is complete. Harvest scheduling with MPIB distribution units should occur immediately to minimize field spoilage.`;
            }
        }

        let growthChart;

        function updateUI() {
            // Update labels corresponding to visual values
            document.getElementById('valL').textContent = `${L} g`;
            document.getElementById('valK').textContent = `${k.toFixed(2)}`;
            document.getElementById('valX0').textContent = `Week ${x0}`;

            const computedData = generateChartData();
            
            // Re-render chart dataset values dynamically
            growthChart.data.labels = computedData.labels;
            growthChart.data.datasets[0].data = computedData.dataFx;
            growthChart.data.datasets[1].data = computedData.dataFprime;
            growthChart.update('active'); // Maintain responsive animations

            // Pull active selected indices and run specific derivative calculus updates
            const currentX = selectedIndex * 0.5;
            const currentFx = f(currentX, L, k, x0);
            const currentFprime = fPrime(currentX, L, k, x0);

            // Print on screen panel values
            document.getElementById('calcWeekTitle').textContent = `WEEK ${currentX.toFixed(1)}`;
            document.getElementById('resultWeight').textContent = currentFx.toFixed(1);
            document.getElementById('resultDerivative').textContent = currentFprime.toFixed(1);

            // Generate agricultural layman advice block
            const explanation = getLaymanAssessment(currentX, currentFx, currentFprime, k, x0);
            document.getElementById('laymanExplanation').innerHTML = explanation.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
        }
